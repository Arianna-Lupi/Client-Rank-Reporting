export interface DailyMetricRow {
  date: string; // 'YYYY-MM-DD'
  clicks: number;
  impressions: number;
  ctr: number;      // fracción 0-1
  position: number; // posición media; menor es mejor
}

export interface MetricDelta {
  value: number;
  previous: number;
  deltaPct: number | null; // null si previous === 0 (isNew true)
  improved: boolean;       // invertido para position

}
export interface MetricDeltas { clicks: MetricDelta; impressions: MetricDelta; ctr: MetricDelta; position: MetricDelta; }

// Ya invierte position y guarda ÷0. Toma dos DailyMetricRow y devuelve MetricDeltas.
