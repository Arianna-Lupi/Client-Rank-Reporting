lib/site-match.ts:
```typescript
/** Lowercased bare host: scheme, `sc-domain:`, `www.` and trailing slash removed. */
export function normalizeSiteRef(input: string): string;

export type ResolveResult =
  | { kind: 'match'; siteUrl: string }
  | { kind: 'multiple'; candidates: string[] }
  | { kind: 'none' };

/**
 * Resolve a user-typed ref against a list of canonical siteUrls.
 * 1. Exact (case-sensitive) equality on a canonical siteUrl wins outright.
 * 2. Otherwise normalize input and candidates; collect candidates whose
 *    normalized form equals the normalized input. 1 -> match, >1 -> multiple, 0 -> none.
 */
export function resolveSiteRef(input: string, candidates: ReadonlyArray<string>): ResolveResult;
```

lib/clients.ts (additions):
```typescript
/** Minimal write surface — lets tests inject a fake without live Upstash. */
export interface ActiveClientWriter {
  sadd(key: string, ...members: string[]): Promise<number>;
  srem(key: string, ...members: string[]): Promise<number>;
}

/** SADD clients:active <siteUrl>. Returns true if newly added (sadd === 1). */
export function addClient(siteUrl: string, writer?: ActiveClientWriter): Promise<boolean>;

/** SREM clients:active <siteUrl>. Returns true if it was present (srem === 1). */
export function removeClient(siteUrl: string, writer?: ActiveClientWriter): Promise<boolean>;
```
